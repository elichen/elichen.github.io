cmake_minimum_required(VERSION 3.15)
project(headless CXX)
# BAAIWorm's SoftCon body (Metaworm/sim) + a headless interact module; no renderer, no CUDA
# cmake -S . -B build -DBAAIWORM=<checkout> -DPython_EXECUTABLE=<venv python>
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
find_package(Eigen3 REQUIRED)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
find_package(pybind11 CONFIG REQUIRED)
add_definitions(-DSOFTCON_DIR="${BAAIWORM}/Metaworm" -DHAVE_CSTDDEF)
file(GLOB sim_srcs ${BAAIWORM}/Metaworm/sim/*.cpp ${BAAIWORM}/Metaworm/sim/util/*.cpp ${BAAIWORM}/Metaworm/sim/util/json/*.cpp ${BAAIWORM}/Metaworm/sim/fem/*.cpp ${BAAIWORM}/Metaworm/sim/fem/Constraint/*.cpp ${BAAIWORM}/Metaworm/sim/fem/Mesh/*.cpp)
add_library(simfem STATIC ${sim_srcs})
target_compile_options(simfem PUBLIC -include cassert)
set_target_properties(simfem PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(simfem PUBLIC ${BAAIWORM}/Metaworm/sim)
target_link_libraries(simfem PUBLIC Eigen3::Eigen)
pybind11_add_module(interact interact_headless.cpp)
target_link_libraries(interact PRIVATE simfem)
